328 void SetTargetBranch(const std::string& branch) {
329 if (branch.empty()) [[unlikely]] {
330 AddError(ErrorKind::kInvalidArgument, "Branch name cannot be empty");
331 return;
332 }
333
334 if (auto ref_it = base().refs.find(branch); ref_it != base().refs.end()) {
335 if (ref_it->second->type() != SnapshotRefType::kBranch) {
336 AddError(ErrorKind::kInvalidArgument,
337 "{} is a tag, not a branch. Tags cannot be targets for "
338 "producing snapshots",
339 branch);
340 return;
341 }
342 }
343
344 target_branch_ = branch;
345 }
346
347 void SetManifestWriter(Executor& executor, int32_t parallelism) {
348 if (parallelism <= 0) [[unlikely]] {
349 AddError(ErrorKind::kInvalidArgument,
350 "Manifest write parallelism must be greater than 0, but was: {}",
351 parallelism);
352 return;
353 }
354
355 write_manifest_executor_ = std::ref(executor);
356 write_manifest_parallelism_ = parallelism;
357 }
