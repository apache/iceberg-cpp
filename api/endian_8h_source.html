36template <typename T>
37concept EndianConvertible = std::is_arithmetic_v<T>;

41template <EndianConvertible T>
42constexpr T ByteSwap(T value) {
43 if constexpr (sizeof(T) <= 1) {
44 return value;
45 } else if constexpr (std::is_integral_v<T>) {
46 return static_cast<T>(
47 internal::ByteSwapUnsigned(static_cast<std::make_unsigned_t<T>>(value)));
48 } else if constexpr (std::is_floating_point_v<T>) {
49 if constexpr (sizeof(T) == sizeof(uint16_t)) {
50 return std::bit_cast<T>(internal::ByteSwapUnsigned(std::bit_cast<uint16_t>(value)));
51 } else if constexpr (sizeof(T) == sizeof(uint32_t)) {
52 return std::bit_cast<T>(internal::ByteSwapUnsigned(std::bit_cast<uint32_t>(value)));
53 } else if constexpr (sizeof(T) == sizeof(uint64_t)) {
54 return std::bit_cast<T>(internal::ByteSwapUnsigned(std::bit_cast<uint64_t>(value)));
55 } else {
56 static_assert(sizeof(T) == 0,
57 "Unsupported floating-point size for endian conversion.");
58 }
59 }
60}
