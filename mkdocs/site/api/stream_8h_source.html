69 Result<std::optional<T>> Next() {
70 if (error_.has_value()) {
71 return std::unexpected(*error_);
72 }
73 if (finished_) {
74 return std::nullopt;
75 }
76
77 auto result = NextImpl();
78 if (!result.has_value()) {
79 error_ = result.error();
80 } else if (!result.value().has_value()) {
81 finished_ = true;
82 }
83 return result;
84 }

87 Result<std::vector<T>> ToVector() {
88 auto collect = [this](auto& values, auto append,
89 auto finish) -> Result<std::vector<T>> {
90 while (true) {
91 auto result = Next();
92 if (!result.has_value()) {
93 return std::unexpected(std::move(result.error()));
94 }
95 auto& value = result.value();
96 if (!value.has_value()) {
97 return finish(values);
98 }
99 append(values, value.value());
100 }
101 };
102
103 if constexpr (!std::is_move_constructible_v<T>) {
104 static_assert(std::is_copy_constructible_v<T>,
105 "Stream::ToVector requires T to be move- or copy-constructible");
106
107 // For strictly copy-only T, collecting directly into a vector can repeatedly copy
108 // previously collected elements during vector growth. Stage values in a deque,
109 // then copy once into an exactly sized vector.
110 std::deque<T> values;
111 return collect(
112 values, [](auto& destination, const T& value) { destination.push_back(value); },
113 [](const auto& source) {
114 return std::vector<T>(source.cbegin(), source.cend());
115 });
116 } else {
117 std::vector<T> values;
118 return collect(
119 values,
120 [](auto& destination, T& value) {
121 destination.push_back(std::move_if_noexcept(value));
122 },
123 [](auto& source) { return std::move(source); });
124 }
125 }
126
127 protected:
132 virtual Result<std::optional<T>> NextImpl() = 0;
133
134 private:
135 bool finished_ = false;
136 std::optional<Error> error_;
137};
