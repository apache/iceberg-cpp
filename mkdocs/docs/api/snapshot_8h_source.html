54ICEBERG_EXPORT constexpr std::string_view ToString(SnapshotRefType type) noexcept {
55 switch (type) {
56 case SnapshotRefType::kBranch:
57 return "branch";
58 case SnapshotRefType::kTag:
59 return "tag";
60 }
61 ::iceberg::unreachable();
62}
64ICEBERG_EXPORT constexpr Result<SnapshotRefType> SnapshotRefTypeFromString(
65 std::string_view str) noexcept {
66 if (str == "branch") return SnapshotRefType::kBranch;
67 if (str == "tag") return SnapshotRefType::kTag;
68 return InvalidArgument("Invalid snapshot reference type: {}", str);
69}
